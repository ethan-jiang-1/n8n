// 环境变量验证器
class EnvironmentValidator {
  private requiredVars: string[];
  private validationRules: Map<string, ValidationRule>;
  
  constructor() {
    this.requiredVars = [
      'N8N_ENCRYPTION_KEY',
      'DB_TYPE'
    ];
    
    this.initializeValidationRules();
  }
  
  public validateEnvironment(): ValidationResult {
    const errors: ValidationError[] = [];
    const warnings: ValidationWarning[] = [];
    
    // 1. 检查必需变量
    for (const varName of this.requiredVars) {
      if (!process.env[varName]) {
        errors.push({
          variable: varName,
          type: 'missing_required',
          message: `Required environment variable ${varName} is not set`
        });
      }
    }
    
    // 2. 验证变量格式和值
    for (const [varName, rule] of this.validationRules) {
      const value = process.env[varName];
      if (value) {
        const validationResult = rule.validate(value);
        if (!validationResult.isValid) {
          errors.push({
            variable: varName,
            type: 'invalid_format',
            message: validationResult.message
          });
        }
      }
    }
    
    // 3. 检查配置冲突
    const conflicts = this.checkConfigurationConflicts();
    errors.push(...conflicts);
    
    // 4. 性能建议
    const suggestions = this.generatePerformanceSuggestions();
    warnings.push(...suggestions);
    
    return {
      isValid: errors.length === 0,
      errors,
      warnings
    };
  }
  
  private initializeValidationRules(): void {
    this.validationRules.set('N8N_PORT', {
      validate: (value: string) => {
        const port = parseInt(value);
        return {
          isValid: port >= 1 && port <= 65535,
          message: 'Port must be between 1 and 65535'
        };
      }
    });
    
    this.validationRules.set('EXECUTIONS_TIMEOUT', {
      validate: (value: string) => {
        const timeout = parseInt(value);
        return {
          isValid: timeout > 0 && timeout <= 7200, // 2 hours max
          message: 'Execution timeout must be between 1 and 7200 seconds'
        };
      }
    });
    
    this.validationRules.set('N8N_ENCRYPTION_KEY', {
      validate: (value: string) => {
        return {
          isValid: value.length >= 32,
          message: 'Encryption key must be at least 32 characters long'
        };
      }
    });
  }
}

// 配置管理器
class ConfigurationManager {
  private config: Map<string, any> = new Map();
  private watchers: Map<string, ConfigWatcher[]> = new Map();
  private validator: EnvironmentValidator;
  
  constructor() {
    this.validator = new EnvironmentValidator();
    this.loadConfiguration();
    this.setupConfigurationWatching();
  }
  
  public getConfig<T>(key: string, defaultValue?: T): T {
    return this.config.get(key) ?? defaultValue;
  }
  
  public setConfig(key: string, value: any): void {
    const oldValue = this.config.get(key);
    this.config.set(key, value);
    
    // 触发配置变更监听器
    this.notifyWatchers(key, value, oldValue);
  }
  
  public watchConfig(
    key: string,
    callback: (newValue: any, oldValue: any) => void
  ): void {
    if (!this.watchers.has(key)) {
      this.watchers.set(key, []);
    }
    
    this.watchers.get(key)!.push({ callback });
  }
  
  private loadConfiguration(): void {
    // 1. 从环境变量加载
    this.loadFromEnvironmentVariables();
    
    // 2. 从配置文件加载
    this.loadFromConfigurationFiles();
    
    // 3. 应用默认值
    this.applyDefaultValues();
    
    // 4. 验证配置
    const validation = this.validator.validateEnvironment();
    if (!validation.isValid) {
      throw new Error(`Configuration validation failed: ${
        validation.errors.map(e => e.message).join(', ')
      }`);
    }
  }
  
  private loadFromEnvironmentVariables(): void {
    // 系统环境变量映射
    const envMappings: Record<string, string> = {
      'server.host': 'N8N_HOST',
      'server.port': 'N8N_PORT',
      'server.protocol': 'N8N_PROTOCOL',
      'execution.timeout': 'EXECUTIONS_TIMEOUT',
      'execution.mode': 'EXECUTIONS_MODE',
      'database.type': 'DB_TYPE',
      'logging.level': 'N8N_LOG_LEVEL',
      'security.encryptionKey': 'N8N_ENCRYPTION_KEY',
      'taskRunner.enabled': 'N8N_RUNNERS_ENABLED',
      'taskRunner.mode': 'N8N_RUNNERS_MODE',
      'performance.cacheEnabled': 'N8N_CACHE_ENABLED'
    };
    
    for (const [configKey, envKey] of Object.entries(envMappings)) {
      const envValue = process.env[envKey];
      if (envValue !== undefined) {
        this.config.set(configKey, this.parseEnvValue(envValue));
      }
    }
  }
  
  private parseEnvValue(value: string): any {
    // 尝试解析为布尔值
    if (value.toLowerCase() === 'true') return true;
    if (value.toLowerCase() === 'false') return false;
    
    // 尝试解析为数字
    const numValue = Number(value);
    if (!isNaN(numValue)) return numValue;
    
    // 尝试解析为 JSON
    try {
      return JSON.parse(value);
    } catch {
      // 返回字符串
      return value;
    }
  }
}

// 配置热重载管理器
class ConfigurationHotReloader {
  private configWatchers: Map<string, FileWatcher> = new Map();
  private reloadCallbacks: Map<string, ReloadCallback[]> = new Map();
  private configurationManager: ConfigurationManager;
  
  constructor(configManager: ConfigurationManager) {
    this.configurationManager = configManager;
    this.setupConfigurationWatching();
  }
  
  public watchConfigurationFile(
    filePath: string,
    callback: ReloadCallback
  ): void {
    if (!this.configWatchers.has(filePath)) {
      const watcher = this.createFileWatcher(filePath);
      this.configWatchers.set(filePath, watcher);
    }
    
    if (!this.reloadCallbacks.has(filePath)) {
      this.reloadCallbacks.set(filePath, []);
    }
    
    this.reloadCallbacks.get(filePath)!.push(callback);
  }
  
  private createFileWatcher(filePath: string): FileWatcher {
    return new FileWatcher(filePath, {
      // 防抖动配置
      debounceTime: 1000, // 1秒
      
      // 监听事件类型
      events: ['change', 'rename'],
      
      // 忽略临时文件
      ignored: /[\/\\]\./,
      
      // 持久化监听
      persistent: true
    }).on('change', async (eventType, filename) => {
      try {
        await this.handleConfigurationChange(filePath, eventType);
      } catch (error) {
        console.error(`Failed to reload configuration from ${filePath}:`, error);
      }
    });
  }
  
  private async handleConfigurationChange(
    filePath: string,
    eventType: string
  ): Promise<void> {
    // 1. 验证文件完整性
    const isValidFile = await this.validateConfigurationFile(filePath);
    if (!isValidFile) {
      console.warn(`Configuration file ${filePath} is not valid, skipping reload`);
      return;
    }
    
    // 2. 备份当前配置
    const currentConfig = this.configurationManager.exportConfiguration();
    
    try {
      // 3. 重新加载配置
      const newConfig = await this.loadConfigurationFromFile(filePath);
      
      // 4. 验证新配置
      const validation = await this.validateConfiguration(newConfig);
      if (!validation.isValid) {
        throw new Error(`Invalid configuration: ${validation.errors.join(', ')}`);
      }
      
      // 5. 应用新配置
      await this.applyConfiguration(newConfig);
      
      // 6. 通知回调函数
      const callbacks = this.reloadCallbacks.get(filePath) || [];
      for (const callback of callbacks) {
        await callback(newConfig, currentConfig);
      }
      
      console.log(`Configuration successfully reloaded from ${filePath}`);
      
    } catch (error) {
      // 配置加载失败，恢复备份
      await this.configurationManager.importConfiguration(currentConfig);
      throw error;
    }
  }
  
  private async applyConfiguration(config: Configuration): Promise<void> {
    // 逐步应用配置变更，确保系统稳定性
    const changeGroups = this.groupConfigurationChanges(config);
    
    for (const group of changeGroups) {
      switch (group.type) {
        case 'low_impact':
          // 低影响配置直接应用
          await this.applyLowImpactChanges(group.changes);
          break;
          
        case 'medium_impact':
          // 中等影响配置需要验证
          await this.applyMediumImpactChanges(group.changes);
          break;
          
        case 'high_impact':
          // 高影响配置需要重启相关服务
          await this.applyHighImpactChanges(group.changes);
          break;
      }
    }
  }
}

class ConfigurationHealthChecker {
  private healthChecks: Map<string, HealthCheck> = new Map();
  private lastCheckResults: Map<string, HealthCheckResult> = new Map();
  
  constructor() {
    this.initializeHealthChecks();
  }
  
  public async performHealthCheck(): Promise<OverallHealthResult> {
    const results = new Map<string, HealthCheckResult>();
    const startTime = Date.now();
    
    // 并行执行所有健康检查
    const checkPromises = Array.from(this.healthChecks.entries()).map(
      async ([name, check]) => {
        try {
          const result = await this.executeHealthCheck(name, check);
          results.set(name, result);
        } catch (error) {
          results.set(name, {
            status: 'unhealthy',
            message: error.message,
            timestamp: Date.now(),
            duration: 0
          });
        }
      }
    );
    
    await Promise.all(checkPromises);
    
    const endTime = Date.now();
    const overallStatus = this.calculateOverallStatus(results);
    
    return {
      status: overallStatus,
      checks: Object.fromEntries(results),
      duration: endTime - startTime,
      timestamp: endTime
    };
  }
  
  private initializeHealthChecks(): void {
    // 数据库连接检查
    this.healthChecks.set('database_connection', {
      name: 'Database Connection',
      timeout: 5000,
      execute: async () => {
        const dbConfig = this.getDbConfig();
        await this.testDatabaseConnection(dbConfig);
        return { healthy: true, message: 'Database connection OK' };
      }
    });
    
    // Task Runner 健康检查
    this.healthChecks.set('task_runner_health', {
      name: 'Task Runner Health',
      timeout: 3000,
      execute: async () => {
        if (!process.env.N8N_RUNNERS_ENABLED) {
          return { healthy: true, message: 'Task Runner disabled' };
        }
        
        const runnerStatus = await this.checkTaskRunnerStatus();
        return {
          healthy: runnerStatus.healthy,
          message: runnerStatus.message,
          metrics: runnerStatus.metrics
        };
      }
    });
    
    // 内存使用检查
    this.healthChecks.set('memory_usage', {
      name: 'Memory Usage',
      timeout: 1000,
      execute: async () => {
        const memUsage = process.memoryUsage();
        const memThreshold = parseInt(process.env.N8N_MEMORY_THRESHOLD || '1073741824'); // 1GB
        
        const isHealthy = memUsage.heapUsed < memThreshold;
        return {
          healthy: isHealthy,
          message: isHealthy ? 'Memory usage normal' : 'Memory usage high',
          metrics: {
            heapUsed: memUsage.heapUsed,
            heapTotal: memUsage.heapTotal,
            external: memUsage.external,
            rss: memUsage.rss
          }
        };
      }
    });
    
    // 配置一致性检查
    this.healthChecks.set('configuration_consistency', {
      name: 'Configuration Consistency',
      timeout: 2000,
      execute: async () => {
        const inconsistencies = await this.findConfigurationInconsistencies();
        return {
          healthy: inconsistencies.length === 0,
          message: inconsistencies.length === 0 
            ? 'Configuration is consistent'
            : `Found ${inconsistencies.length} inconsistencies`,
          details: inconsistencies
        };
      }
    });
  }
  
  private async findConfigurationInconsistencies(): Promise<ConfigInconsistency[]> {
    const inconsistencies: ConfigInconsistency[] = [];
    
    // 检查执行模式和数据库类型的兼容性
    const executionMode = process.env.EXECUTIONS_MODE;
    const dbType = process.env.DB_TYPE;
    
    if (executionMode === 'queue' && dbType === 'sqlite') {
      inconsistencies.push({
        type: 'compatibility',
        severity: 'warning',
        message: 'Queue mode is not recommended with SQLite database',
        suggestion: 'Consider using MySQL or PostgreSQL for queue mode'
      });
    }
    
    // 检查安全配置
    if (process.env.NODE_ENV === 'production') {
      if (!process.env.N8N_ENCRYPTION_KEY) {
        inconsistencies.push({
          type: 'security',
          severity: 'critical',
          message: 'Encryption key not set in production',
          suggestion: 'Set N8N_ENCRYPTION_KEY environment variable'
        });
      }
      
      if (process.env.N8N_BASIC_AUTH_ACTIVE !== 'true' && 
          process.env.N8N_JWT_AUTH_ACTIVE !== 'true') {
        inconsistencies.push({
          type: 'security',
          severity: 'high',
          message: 'No authentication enabled in production',
          suggestion: 'Enable basic auth or JWT authentication'
        });
      }
    }
    
    return inconsistencies;
  }
}

class ConfigurationPerformanceAnalyzer {
  private performanceMetrics: Map<string, PerformanceMetric[]> = new Map();
  private configurationImpactAnalyzer: ConfigurationImpactAnalyzer;
  
  public async analyzeConfigurationPerformance(): Promise<PerformanceAnalysisReport> {
    const analysis: PerformanceAnalysisReport = {

      categories: {}
    };
    
    // 1. 分析执行配置性能
    analysis.categories.execution = await this.analyzeExecutionConfiguration();

    // 4. 分析安全配置性能

    return analysis;
  }
  
  private async analyzeExecutionConfiguration(): Promise<CategoryPerformanceAnalysis> {
    const config = {
      timeout: parseInt(process.env.EXECUTIONS_TIMEOUT || '120'),
      mode: process.env.EXECUTIONS_MODE || 'regular',
      process: process.env.EXECUTIONS_PROCESS || 'main'
    };
    
    const analysis: CategoryPerformanceAnalysis = {
      score: 100,
      issues: [],
      recommendations: []
    };
    
    // 执行超时分析
    if (config.timeout < 30) {
      analysis.score -= 20;
      analysis.issues.push({
        type: 'configuration',
        severity: 'medium',
        message: 'Execution timeout is very low',
        impact: 'May cause frequent timeouts for complex workflows'
      });
      
      analysis.recommendations.push({
        type: 'increase_timeout',
        priority: 'medium',
        message: 'Consider increasing EXECUTIONS_TIMEOUT to at least 120 seconds',
        expectedImprovement: 15
      });
    }
    
    // 执行模式分析
    if (config.mode === 'regular' && this.getWorkflowComplexity() > 0.8) {
      analysis.score -= 15;
      analysis.recommendations.push({
        type: 'enable_queue_mode',
        priority: 'high',
        message: 'Enable queue mode for better scalability with complex workflows',
        expectedImprovement: 25
      });
    }
    
    return analysis;
  }
  
  private async analyzeDatabaseConfiguration(): Promise<CategoryPerformanceAnalysis> {
    const dbType = process.env.DB_TYPE || 'sqlite';
    const analysis: CategoryPerformanceAnalysis = {
      score: 100,
      issues: [],
      recommendations: []
    };
    
    // SQLite 性能分析
    if (dbType === 'sqlite') {
      const dbSize = await this.getDatabaseSize();
      if (dbSize > 100 * 1024 * 1024) { // 100MB
        analysis.score -= 30;
        analysis.issues.push({
          type: 'scalability',
          severity: 'high',
          message: 'SQLite database is large',
          impact: 'Performance degradation with large databases'
        });
        
        analysis.recommendations.push({
          type: 'migrate_database',
          priority: 'high',
          message: 'Consider migrating to PostgreSQL or MySQL for better performance',
          expectedImprovement: 40
        });
      }
    }
    
    // 数据库连接池分析
    if (dbType !== 'sqlite') {
      const poolSize = this.getDatabasePoolSize();
      const recommendedPoolSize = this.calculateRecommendedPoolSize();
      
      if (poolSize < recommendedPoolSize) {
        analysis.score -= 10;
        analysis.recommendations.push({
          type: 'increase_pool_size',
          priority: 'medium',
          message: `Increase database connection pool size to ${recommendedPoolSize}`,
          expectedImprovement: 15
        });
      }
    }
    
    return analysis;
  }
  
  public generateOptimizationReport(): ConfigurationOptimizationReport {
    const report: ConfigurationOptimizationReport = {
      timestamp: Date.now(),
      categories: [],
      overallImpact: 0,
      implementationComplexity: 'medium'
    };
    
    // 生成各类别的优化建议
    report.categories = [
      this.generateExecutionOptimizations(),
      this.generatePerformanceOptimizations(),
      this.generateSecurityOptimizations(),
      this.generateScalabilityOptimizations()
    ];
    
    // 计算总体影响
    report.overallImpact = report.categories.reduce(
      (sum, category) => sum + category.expectedImprovement,
      0
    ) / report.categories.length;
    
    // 确定实施复杂度
    const hasHighComplexityChanges = report.categories.some(
      category => category.complexity === 'high'
    );
    
    report.implementationComplexity = hasHighComplexityChanges ? 'high' : 'medium';
    
    return report;
  }
}
