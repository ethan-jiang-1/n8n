## 1. 常见启动问题

### 1.1 端口冲突问题

#### 问题现象
```bash
Error: listen EADDRINUSE: address already in use :::5678
```

#### 解决方案
```bash
# 方法 1: 查找并杀死占用进程
lsof -i :5678
kill -9 <PID>

# 方法 2: 使用不同端口
N8N_PORT=8080 pnpm dev

# 方法 3: 批量清理 Node.js 进程
pkill -f "node.*n8n"
pkill -f "node.*packages"

# 方法 4: 系统级端口检查 (macOS)
sudo lsof -i :5678
sudo kill -9 <PID>

# 方法 5: 自动杀死进程并启动
./scripts/kill-and-start.sh
```

创建自动化脚本：
```bash
#!/bin/bash
# scripts/kill-and-start.sh

echo "Stopping existing n8n processes..."
pkill -f "node.*n8n" || true
pkill -f "pnpm.*dev" || true

echo "Waiting for processes to stop..."
sleep 3

echo "Starting n8n development server..."
pnpm dev
```

#### 问题现象

```bash
#!/bin/bash
# scripts/monitor-memory.sh

while true; do
  echo "=== $(date) ==="
  
  # 系统内存使用
  echo "System Memory:"
  free -h
  
  # Node.js 进程内存使用
  echo "Node.js Processes:"
  ps aux | grep node | grep -v grep | awk '{print $2, $4, $6, $11}' | column -t

  echo "n8n Processes:"
  ps aux | grep n8n | grep -v grep
  
  echo "=================="
  sleep 30
done
```

# 监控脚本
#!/bin/bash
# scripts/monitor-cpu.sh

while true; do
  echo "=== CPU Usage $(date) ==="
  
  # 整体 CPU 使用率

  # Node.js 进程 CPU 使用率

// packages/cli/src/performance/PerformanceMonitor.ts
export class PerformanceMonitor {
  private static instance: PerformanceMonitor;
  
  static getInstance(): PerformanceMonitor {
    if (!this.instance) {
      this.instance = new PerformanceMonitor();
    }
    return this.instance;
  }
  
  logExecutionTime<T>(
    operation: string,
    fn: () => Promise<T>
  ): Promise<T> {
    const startTime = Date.now();
    
    return fn().finally(() => {
      const duration = Date.now() - startTime;
      console.log(`${operation} took ${duration}ms`);
      
      if (duration > 1000) {
        console.warn(`Slow operation detected: ${operation} (${duration}ms)`);
      }
    });
  }
  
  logMemoryUsage(label: string): void {
    const usage = process.memoryUsage();
    console.log(`${label} Memory:`, {
      rss: `${Math.round(usage.rss / 1024 / 1024)}MB`,
      heapTotal: `${Math.round(usage.heapTotal / 1024 / 1024)}MB`,
      heapUsed: `${Math.round(usage.heapUsed / 1024 / 1024)}MB`
    });
  }
}

// 使用示例
const monitor = PerformanceMonitor.getInstance();

await monitor.logExecutionTime('workflow-execution', async () => {
  return await workflowRunner.run(workflow);
});

monitor.logMemoryUsage('After workflow execution');

## 9. 内存泄漏检测

### 9.1 内存泄漏监控

```typescript
// packages/cli/src/utils/MemoryLeakDetector.ts
export class MemoryLeakDetector {
  private baselineMemory: NodeJS.MemoryUsage;
  private checkInterval: NodeJS.Timer;
  private measurements: Array<{ timestamp: number; memory: NodeJS.MemoryUsage }> = [];
  
  constructor(private threshold: number = 100 * 1024 * 1024) { // 100MB
    this.baselineMemory = process.memoryUsage();
  }
  
  startMonitoring(intervalMs: number = 30000): void {
    this.checkInterval = setInterval(() => {
      this.checkMemoryUsage();
    }, intervalMs);
  }
  
  stopMonitoring(): void {
    if (this.checkInterval) {
      clearInterval(this.checkInterval);
    }
  }
  
  private checkMemoryUsage(): void {
    const currentMemory = process.memoryUsage();
    const memoryIncrease = currentMemory.heapUsed - this.baselineMemory.heapUsed;
    
    this.measurements.push({
      timestamp: Date.now(),
      memory: currentMemory
    });
    
    // 保留最近100个测量值
    if (this.measurements.length > 100) {
      this.measurements.shift();
    }
    
    if (memoryIncrease > this.threshold) {
      console.warn('Potential memory leak detected:', {
        baseline: this.formatBytes(this.baselineMemory.heapUsed),
        current: this.formatBytes(currentMemory.heapUsed),
        increase: this.formatBytes(memoryIncrease)
      });
      
      this.generateMemoryReport();
    }
  }
  
  private formatBytes(bytes: number): string {
    return `${Math.round(bytes / 1024 / 1024)}MB`;
  }
  
  private generateMemoryReport(): void {
    const recent = this.measurements.slice(-10);
    const trend = this.calculateMemoryTrend(recent);
    
    console.log('Memory Usage Trend:', {
      measurements: recent.length,
      trend: trend > 0 ? 'increasing' : 'decreasing',
      avgIncrease: this.formatBytes(trend)
    });
  }
  
  private calculateMemoryTrend(measurements: typeof this.measurements): number {
    if (measurements.length < 2) return 0;
    
    const first = measurements[0];
    const last = measurements[measurements.length - 1];
    
    return (last.memory.heapUsed - first.memory.heapUsed) / measurements.length;
  }
}

// 使用示例
const memoryDetector = new MemoryLeakDetector();
memoryDetector.startMonitoring(30000); // 每30秒检查一次

// 在应用关闭时停止监控
process.on('SIGTERM', () => {
  memoryDetector.stopMonitoring();
});
```

## 11. 自动化监控和告警

### 11.1 健康检查脚本

```bash
#!/bin/bash
# scripts/health-check.sh

# 配置
MAX_MEMORY_MB=2048
MAX_CPU_PERCENT=80
API_ENDPOINT="http://localhost:5678/rest/health"

# 检查 API 健康状态
check_api_health() {
  local response=$(curl -s -o /dev/null -w "%{http_code}" "$API_ENDPOINT")
  
  if [ "$response" != "200" ]; then
    echo "❌ API health check failed (HTTP $response)"
    return 1
  else
    echo "✅ API health check passed"
    return 0
  fi
}

# 检查内存使用
check_memory_usage() {
  local memory_mb=$(ps -o pid,rss -p $(pgrep -f "node.*n8n") | awk 'NR>1 {sum+=$2} END {print sum/1024}')
  
  if (( $(echo "$memory_mb > $MAX_MEMORY_MB" | bc -l) )); then
    echo "❌ High memory usage: ${memory_mb}MB (threshold: ${MAX_MEMORY_MB}MB)"
    return 1
  else
    echo "✅ Memory usage normal: ${memory_mb}MB"
    return 0
  fi
}

# 检查 CPU 使用
check_cpu_usage() {
  local cpu_percent=$(ps -o pid,pcpu -p $(pgrep -f "node.*n8n") | awk 'NR>1 {sum+=$2} END {print sum}')
  
  if (( $(echo "$cpu_percent > $MAX_CPU_PERCENT" | bc -l) )); then
    echo "❌ High CPU usage: ${cpu_percent}% (threshold: ${MAX_CPU_PERCENT}%)"
    return 1
  else
    echo "✅ CPU usage normal: ${cpu_percent}%"
    return 0
  fi
}

# 主检查流程
echo "=== n8n Health Check $(date) ==="

api_ok=0
memory_ok=0
cpu_ok=0

check_api_health && api_ok=1
check_memory_usage && memory_ok=1
check_cpu_usage && cpu_ok=1

# 综合结果
if [ $api_ok -eq 1 ] && [ $memory_ok -eq 1 ] && [ $cpu_ok -eq 1 ]; then
  echo "✅ All health checks passed"
  exit 0
else
  echo "❌ Health check failed"
  exit 1
fi
```

### 11.2 自动重启脚本

```bash
#!/bin/bash
# scripts/auto-restart.sh

LOG_FILE="/tmp/n8n-monitor.log"
MAX_RESTARTS=5
RESTART_COUNT=0

log_message() {
  echo "$(date): $1" | tee -a "$LOG_FILE"
}

restart_n8n() {
  log_message "Restarting n8n (attempt $((RESTART_COUNT + 1))/$MAX_RESTARTS)"
  
  # 停止现有进程
  pkill -f "node.*n8n"
  sleep 5
  
  # 启动新进程
  cd /path/to/n8n
  pnpm start &
  
  RESTART_COUNT=$((RESTART_COUNT + 1))
  sleep 30  # 等待启动完成
}

# 主监控循环
while true; do
  if ./scripts/health-check.sh; then
    RESTART_COUNT=0  # 重置重启计数
  else
    if [ $RESTART_COUNT -lt $MAX_RESTARTS ]; then
      restart_n8n
    else
      log_message "Maximum restart attempts reached. Manual intervention required."
      # 发送告警通知
      echo "n8n requires manual intervention" | mail -s "n8n Alert" admin@example.com
      exit 1
    fi
  fi
  
  sleep 60  # 每分钟检查一次
done
```
